1. Write a program to take marks of a student and print the grade:(using user input)
import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int mark = sc.nextInt();
        System.out.println("Enter Marks: ");
        if (mark>90){
            System.out.println("A grade");
        }
        else if (mark>80 && mark<=90){
            System.out.println("B grade");
        }
        else if (mark>=70 && mark<=80){
            System.out.println("C grade");
        }
        else{
            System.out.println("D grade");
        }
    }
}

5.  Write a program to find the biggest number of three(2378,675,8907)
import java.util.Scanner;
public class ThreeLargest {
    public static void main(String[] args) {
        Scanner reader = new Scanner(System.in);
        System.out.println("Enter a number 1: ");
        int a = reader.nextInt();
        System.out.println("Enter a number 2: ");
        int b = reader.nextInt();
        System.out.println("Enter a number 3: ");
        int c = reader.nextInt();
        if(a > b && a > c){
            System.out.println(a + "  is Largest");
        } else if(b > a && b > c){
            System.out.println(b + "  is Largest");
        }
        else{
            System.out.println(c + "  is Largest");
        }
    }
}
6.  Number swapping with third variable
import java.util.Scanner;
public class Main{
    public static void main(String[]args){
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter a value: ");
        int a = sc.nextInt();
        System.out.println("Enter b value: ");
        int b = sc.nextInt();
        int temp;
        temp=a;
        a=b;
        b=temp;
        System.out.println(a);
        System.out.println(b);
    }
}
